--assistente
create or replace function assitente_incremet_id_func() 
returns trigger as 
$$
declare
    max_id integer = 0;
begin
    if ((select max(id_assistente) from assistente) is not null) then
       max_id = (select max(id_assistente) from assistente);
    end if;
    new.id_assistente = max_id + 1;
    return new;
end;
$$
language plpgsql;

create or replace trigger assitente_incremet_id
before insert on assistente
for each row execute procedure assitente_incremet_id_func()
